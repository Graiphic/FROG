cmake_minimum_required(VERSION 3.24)
project(FrogReferenceCBridge VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "This finite reference bridge targets Windows x64 and MSVC C17.")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The reference bridge requires a 64-bit host toolchain.")
endif()
set(FROG_BRIDGE_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../sdk/ecosystem" CACHE PATH "Public FROG bridge source kit")
if(NOT TARGET frog-bridge-protocol)
  add_subdirectory("${FROG_BRIDGE_SDK_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/sdk" EXCLUDE_FROM_ALL)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/profile.json" REFERENCE_C_PROFILE)
configure_file(src/Profile.hpp.in "${CMAKE_CURRENT_BINARY_DIR}/generated/Profile.hpp" @ONLY)
add_executable(frog-reference-c-bridge src/main.cpp src/Fir.cpp src/Generate.cpp src/Process.cpp)
target_compile_features(frog-reference-c-bridge PRIVATE cxx_std_20)
target_include_directories(frog-reference-c-bridge PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(frog-reference-c-bridge PRIVATE frog-bridge-protocol bcrypt version)
target_compile_options(frog-reference-c-bridge PRIVATE /utf-8 /W4 /EHsc /bigobj /FS)
# Inherit CMAKE_MSVC_RUNTIME_LIBRARY, including a standalone kit's explicit /MT.
# With no override CMake's ordinary /MD matches the canonical public SDK owner.
install(TARGETS frog-reference-c-bridge RUNTIME DESTINATION bin)
install(FILES profile.json DESTINATION share/frog/reference-c)
