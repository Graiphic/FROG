cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(FrogBridgeSdk VERSION 0.1.0 LANGUAGES C CXX)
    include(CTest)
endif()
set(FROG_BRIDGE_FRONTEND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../Implementations/Frontend" CACHE PATH
    "Exact public Frontend source owner used for the neutral JSON and SHA256 implementation")
if(NOT TARGET frog-frontend-production)
    add_subdirectory("${FROG_BRIDGE_FRONTEND_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/frontend" EXCLUDE_FROM_ALL)
endif()
add_library(frog-bridge-protocol STATIC src/BridgeProtocol.cpp src/BridgeJson.cpp)
target_compile_features(frog-bridge-protocol PUBLIC cxx_std_20)
target_include_directories(frog-bridge-protocol PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
    PRIVATE "${FROG_BRIDGE_FRONTEND_ROOT}/src")
target_link_libraries(frog-bridge-protocol PRIVATE frog-frontend-production)
if(MSVC)
    target_compile_options(frog-bridge-protocol PRIVATE /utf-8 /W4 /FS /bigobj)
endif()
add_executable(frog-bridge-protocol-smoke EXCLUDE_FROM_ALL tests/bridge_protocol_smoke.cpp)
target_include_directories(frog-bridge-protocol-smoke PRIVATE "${FROG_BRIDGE_FRONTEND_ROOT}/src")
target_link_libraries(frog-bridge-protocol-smoke PRIVATE frog-bridge-protocol frog-frontend frog-frontend-production)
add_test(NAME frog-bridge-protocol-smoke COMMAND frog-bridge-protocol-smoke
    "${CMAKE_CURRENT_SOURCE_DIR}/fixtures" "${FROG_BRIDGE_FRONTEND_ROOT}/tests/fixtures")
set_tests_properties(frog-bridge-protocol-smoke PROPERTIES LABELS "sdk;bridge;protocol;frontend;bounded" TIMEOUT 60)
add_executable(frog-bridge-json-smoke EXCLUDE_FROM_ALL tests/bridge_json_smoke.cpp)
target_link_libraries(frog-bridge-json-smoke PRIVATE frog-bridge-protocol)
add_test(NAME frog-bridge-json-smoke COMMAND frog-bridge-json-smoke)
set_tests_properties(frog-bridge-json-smoke PROPERTIES LABELS "sdk;bridge;json;exact;bounded" TIMEOUT 30)
